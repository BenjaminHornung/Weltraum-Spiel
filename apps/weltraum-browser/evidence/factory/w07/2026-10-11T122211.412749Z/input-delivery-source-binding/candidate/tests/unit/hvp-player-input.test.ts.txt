const nativeFactoryIntent = (session: HvpPhysicsSession, value: HvpPlayerInput, cancel = false): boolean => {
  const port = session as HvpPhysicsSession & { factoryInput?: (input: HvpPlayerInput, cancel: boolean) => boolean };
  if (port.factoryInput) return port.factoryInput(value, cancel);
  session.input(value); return true;
};

const inputHarness = async (factory = true) => {
  const native = await createHvpPhysicsSession([], { x: 10, y: 4, z: 10 }, undefined,
    { spawn: { x: 0, y: 2, z: 0 }, coverage: [{ minX: -16, minZ: -16, maxX: 16, maxZ: 16 }] });
  const doc = Object.assign(new EventTarget(), { body: new Element(), hidden: false, pointerLockElement: null as Element | null,
    createElement: () => new Element(), exitPointerLock: () => { doc.pointerLockElement = null; doc.dispatchEvent(new Event("pointerlockchange")); } });
  const canvas = Object.assign(new Element(), { requestPointerLock: async () => { doc.pointerLockElement = canvas; doc.dispatchEvent(new Event("pointerlockchange")); } });
  const win = new EventTarget(); const intents: HvpPlayerInput[] = []; let blocked = false;
  const physics = { read: () => native.read(), setPlayerInput: (value: HvpPlayerInput) => { intents.push({ ...value }); native.input(value); },
    setCameraOffset: () => {}, command: async (kind: string) => { if (kind === "Play") native.play(); if (kind === "Pause") native.pause(); } } as unknown as HvpPhysicsClient;
  // The ninth opt-in is deliberately ignored by the old implementation, making
  // the initial RED a collected business assertion rather than an import failure.
  const create = createHvpPlayerInput as unknown as (...args: unknown[]) => ReturnType<typeof createHvpPlayerInput>;
  const input = create(canvas, new PerspectiveCamera(), physics, doc, win, () => {}, { confirm() {}, select() {} }, () => blocked, factory);
  cleanup.push(() => { input.dispose(); native.dispose(); });
  input.start(); await turn();
  const key = (type: string, code: string, repeat = false) => win.dispatchEvent(Object.assign(new Event(type, { cancelable: true }), { code, repeat }));
  return { input, native, intents, key, win, block: () => { blocked = true; } };
};

class NativeTransport {
  static current: NativeTransport;
  onmessage: ((event: MessageEvent<HvpPhysicsReply>) => void) | null = null;
  onerror = null; onmessageerror = null; terminated = false;
  messages: HvpPhysicsMessage[] = []; reads: HvpPhysicsMessage[] = []; session!: HvpPhysicsSession;
  supportedFixture = false;
  holdInputAcks=false;inputAcks:number[]=[];
  constructor() { NativeTransport.current = this; }
  postMessage(message: HvpPhysicsMessage) {
    this.messages.push(structuredClone(message));
    void (async () => {
      if (message.kind === "Initialize") this.session = await createHvpPhysicsSession(this.supportedFixture ? [supportedGround()]
        : message.sectors, message.spawn, message.gravity, this.supportedFixture ? {
        spawn: { x: 0, y: 0.92, z: 0 }, coverage: [{ minX: -4, minZ: -4, maxX: 4, maxZ: 4 }]
      } : message.player, message.inertiaSpawn, message.branchSpawn, message.sessionId, undefined, message.branchKind, message.factoryEnabled);
      else if (message.kind === "Read") { if (message.input) this.session.input(message.input); this.reads.push(message); return; }
      else if(message.kind==="FactoryInput"){
        this.session.factoryInput(message.input,message.cancelPendingJump);
        if(this.holdInputAcks)this.inputAcks.push(message.id);else this.onmessage?.(new MessageEvent("message",{data:{id:message.id}}));
        return;
      }
      else if (message.kind === "Pause") this.session.pause();
      else if (message.kind === "Play") this.session.play();
      else if (message.kind === "PrepareFactoryFacility") this.session.prepareFactoryFacility(message.request);
      else if (message.kind === "CommitFactoryFacility") this.session.commitFactoryFacility(message.transactionId);
      else if (message.kind === "RollbackFactoryFacility") this.session.rollbackFactoryFacility(message.transactionId);
      else if (message.kind === "Dispose") this.session.dispose();
      this.reply(message.id);
    })().catch(error => { this.onmessage?.({ data: { id: message.id, rejected: String(error) } } as MessageEvent<HvpPhysicsReply>); });
  }
  reply(id: number) { this.onmessage?.(new MessageEvent<HvpPhysicsReply>("message", { data: { id, snapshot: this.session.read(), clock: { timers: 0, maxTimerGapMs: 0, maxAdvanceMs: 0,
    maxHandlerMs: 0, lastCommand: "explicit-native-transport-fixture", lastHandlerMs: 0, delayedCallbacks: [] } } })); }
  release() { const message = this.reads.shift(); if (!message) throw new Error("Missing held test Read"); this.reply(message.id); }
  terminate() { this.terminated = true; this.session?.dispose(); }
}

describe("W07 actual Native Factory input safety boundaries", () => {
  it("neutralizes a previously admitted level and pending jump on Native hold entry, without changing pose ticks or velocity", async () => {
    const { worker } = await supportedClientHarness(); const native = worker.session;
    native.input({ ...zero, x: 1, jump: true }); const before = native.read();
    native.prepareTerrain("factory_input_hold", 0, [{ index: 0, mesh: supportedGround() }]);
    const held = native.read();
    expect(held).toMatchObject({ status: "Paused", terrainTransaction: "PreparedHeld" });
    expect(held.ticks).toBe(before.ticks); expect(held.player!.position).toEqual(before.player!.position);
    expect(held.player!.velocityY).toBe(before.player!.velocityY);
    native.commitTerrain("factory_input_hold"); native.finalizeTerrain("factory_input_hold");
    expect(native.read().status).toBe("Running");
    for (let step = 0; step < 6; step++) native.advance(1 / 60);
    expect.soft(native.read().player!.position.x).toBeCloseTo(before.player!.position.x, 6);
    expect(native.read().player!.jumpCount).toBe(before.player!.jumpCount);
    expect(native.read().ticks).toBe(before.ticks + 6);
  });

  it("rejects new movement and jump during an actual prepared Native hold without replay after finalize", async () => {
    const { worker } = await supportedClientHarness(); const native = worker.session; const before = native.read();
    native.prepareTerrain("factory_rejected_input", 0, [{ index: 0, mesh: supportedGround() }]);
    expect.soft(nativeFactoryIntent(native, { ...zero, x: 1, jump: true })).toBe(false);
    native.commitTerrain("factory_rejected_input"); native.finalizeTerrain("factory_rejected_input");
    for (let step = 0; step < 6; step++) native.advance(1 / 60);
    expect.soft(native.read().player!.position.x).toBeCloseTo(before.player!.position.x, 6);
    expect(native.read().player!.jumpCount).toBe(before.player!.jumpCount);
  });

  it("ordinary release preserves one native jump but explicit cancellation clears it before the Native step", async () => {
    const first = await supportedClientHarness();
    nativeFactoryIntent(first.worker.session, { ...zero, jump: true }); nativeFactoryIntent(first.worker.session, zero);
    first.worker.session.advance(1 / 60); expect(first.worker.session.read().player!.jumpCount).toBe(1);
    const second = await supportedClientHarness(); const before = second.worker.session.read();
    nativeFactoryIntent(second.worker.session, { ...zero, jump: true }); nativeFactoryIntent(second.worker.session, zero, true);
    expect(second.worker.session.read().ticks).toBe(before.ticks);
    expect(second.worker.session.read().player!.position).toEqual(before.player!.position);
    second.worker.session.advance(1 / 60); expect(second.worker.session.read().player!.jumpCount).toBe(0);
  });

  it("rejects a Factory-only intent on a real ordinary Legacy Native session", async () => {
    const { client, worker } = await clientHarness(false, true); await client.command("Play");
    for (let step = 0; step < 10; step++) worker.session.advance(1 / 60);
    const before = worker.session.read(); expect(before.player!.grounded).toBe(true);
    expect.soft(nativeFactoryIntent(worker.session, { ...zero, x: 1 })).toBe(false);
    for (let step = 0; step < 6; step++) worker.session.advance(1 / 60);
    expect(worker.session.read().player!.position.x).toBeCloseTo(before.player!.position.x, 6);
  });
