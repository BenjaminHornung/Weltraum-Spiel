describe("W07 Factory event intent without render authority", () => {
  it("delivers normal Factory KeyD without any RAF and releases it before a frame", async () => {
    const h = await inputHarness(); const ticks = h.native.read().ticks;
    h.key("keydown", "KeyD"); expect(h.intents).toHaveLength(1); expect(h.intents[0]!.x).toBeCloseTo(-1);
    h.key("keyup", "KeyD"); expect(h.intents.at(-1)).toMatchObject(zero); expect(h.native.read().ticks).toBe(ticks);
  });
  it("stages one Space edge, preserves it through ordinary release, and never stages it again on RAF", async () => {
    const h = await inputHarness(); h.key("keydown", "Space"); expect(h.intents.at(-1)?.jump).toBe(true);
    h.key("keydown", "Space", true); h.key("keyup", "Space"); h.input.update();
    expect(h.intents.filter(value => value.jump)).toHaveLength(1);
  });
  it.each(["Escape", "blur", "hold"])("neutralizes staged movement and jump at %s without another RAF", async kind => {
    const h = await inputHarness(); h.key("keydown", "KeyD"); h.key("keydown", "Space"); expect(h.intents.at(-1)?.jump).toBe(true);
    if (kind === "Escape") h.key("keydown", "Escape"); else if (kind === "blur") h.win.dispatchEvent(new Event("blur")); else { h.block(); h.key("keyup", "KeyD"); }
    expect(h.intents.at(-1)).toMatchObject(zero); expect(h.input.active).toBe(false);
  });
  it("keeps the unextended Legacy input deferred to the existing RAF path", async () => {
    const h = await inputHarness(false); h.key("keydown", "KeyD"); expect(h.intents).toEqual([]); h.input.update(); expect(h.intents.at(-1)!.x).toBeCloseTo(-1);
  });

class NativeTransport {
  static current: NativeTransport;
  onmessage: ((event: MessageEvent<HvpPhysicsReply>) => void) | null = null;
  onerror = null; onmessageerror = null; terminated = false;
  messages: HvpPhysicsMessage[] = []; reads: HvpPhysicsMessage[] = []; session!: HvpPhysicsSession;
  constructor() { NativeTransport.current = this; }
  postMessage(message: HvpPhysicsMessage) {
    this.messages.push(structuredClone(message));
    void (async () => {
      if (message.kind === "Initialize") this.session = await createHvpPhysicsSession(message.sectors, message.spawn, message.gravity, message.player,
        message.inertiaSpawn, message.branchSpawn, message.sessionId, undefined, message.branchKind, message.factoryEnabled);
      else if (message.kind === "Read") { if (message.input) this.session.input(message.input); this.reads.push(message); return; }
      else if (message.kind === "Pause") this.session.pause();
      else if (message.kind === "Play") this.session.play();
      else if (message.kind === "PrepareFactoryFacility") this.session.prepareFactoryFacility(message.request);
      else if (message.kind === "CommitFactoryFacility") this.session.commitFactoryFacility(message.transactionId);
      else if (message.kind === "RollbackFactoryFacility") this.session.rollbackFactoryFacility(message.transactionId);
      else if (message.kind === "Dispose") this.session.dispose();
      this.reply(message.id);
    })().catch(error => { this.onmessage?.({ data: { id: message.id, rejected: String(error) } } as MessageEvent<HvpPhysicsReply>); });
  }
  reply(id: number) { this.onmessage?.(new MessageEvent<HvpPhysicsReply>("message", { data: { id, snapshot: this.session.read(), clock: { timers: 0, maxTimerGapMs: 0, maxAdvanceMs: 0,
    maxHandlerMs: 0, lastCommand: "explicit-native-transport-fixture", lastHandlerMs: 0, delayedCallbacks: [] } } })); }
  release() { const message = this.reads.shift(); if (!message) throw new Error("Missing held test Read"); this.reply(message.id); }
  terminate() { this.terminated = true; this.session?.dispose(); }
}
